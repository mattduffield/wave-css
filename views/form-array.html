        (function () {
          const btn = document.getElementById('useFamily');
          const fa = document.getElementById('guardians');
          if (!btn || !fa || btn.__wired) return;
          btn.__wired = true;
          btn.addEventListener('click', () => {
            fa.dispatchEvent(new CustomEvent('wc-form-array:populate', { detail: { rows: [
              { first_name: 'Rick', last_name: 'Adamo', relationship: 'parent', phone: '555-0102', email: 'rick@x.com', role: 'Father', contact_pref: 'phone', primary: 'no' },
              { first_name: 'Mary', last_name: 'Adamo', relationship: 'grandparent', phone: '555-0103', email: 'mary@x.com', role: 'Grandmother', contact_pref: 'text', primary: 'no' }
            ]}}));
          });
        })();
